32def load_descriptions():
33 descriptions = []
34 for path in sorted(DEVICES_DIR.glob("*/*.json")):
35 with path.open(encoding="utf-8") as source:
36 description = json.load(source)
37 description["_path"] = path
38 validate_description(description)
39 descriptions.append(description)
40
41 registered = [d for d in descriptions if d["implementation"]["manager"]["registered"]]
42 ids = {}
43 symbols = {}
44 for description in registered:
45 device_type = description["deviceType"]
46 type_id = device_type["enumValue"]
47 symbol = device_type["enumSymbol"]
48 if type_id in RESERVED_TYPE_IDS:
49 raise ValueError(f"{description['_path']}: device type ID {type_id} is reserved")
50 if type_id in ids:
51 raise ValueError(f"duplicate device type ID {type_id}: {ids[type_id]} and {description['_path']}")
52 if not symbol:
53 raise ValueError(f"{description['_path']}: registered device requires enumSymbol")
54 if symbol in symbols:
55 raise ValueError(f"duplicate enum symbol {symbol}: {symbols[symbol]} and {description['_path']}")
56 ids[type_id] = description["_path"]
57 symbols[symbol] = description["_path"]
58 return sorted(registered, key=lambda d: d["deviceType"]["enumValue"])

61def validate_description(description):
62 path = description["_path"]
63 for key in ("schemaVersion", "deviceType", "implementation", "configuration", "state", "ui"):
64 if key not in description:
65 raise ValueError(f"{path}: missing {key}")
66 implementation = description["implementation"]
67 manager = implementation.get("manager", {})
68 if manager.get("registered"):
69 factory = implementation.get("factory")
70 if not factory:
71 raise ValueError(f"{path}: registered device requires implementation.factory")
72 unknown = set(factory.get("arguments", [])) - set(FACTORY_ARGUMENTS)
73 if unknown:
74 raise ValueError(f"{path}: unknown factory arguments: {sorted(unknown)}")
75 for implementation_file in (implementation["header"], implementation["source"]):
76 if not (PROJECT_DIR / implementation_file).is_file():
77 raise ValueError(f"{path}: implementation file does not exist: {implementation_file}")
78
79 for area_name in ("configuration", "state"):
80 fields = description[area_name].get("customBytes", description[area_name].get("descriptionCustomBytes", []))
81 occupied = {}
82 for field in fields:
83 if "offset" not in field or "size" not in field:
84 continue
85 for offset in range(field["offset"], field["offset"] + field["size"]):
86 previous = occupied.get(offset)
87 companion = field.get("role", "").startswith("companion-values-for:")
88 if previous and not companion:
89 raise ValueError(f"{path}: overlapping {area_name} byte {offset}: {previous} and {field['name']}")
90 occupied[offset] = field["name"]

255def parse_source(source):
256 single = re.fullmatch(r"customBytes\‍[(\d+)\‍]", source)
257 if single:
258 return int(single.group(1)), int(single.group(1))
259 span = re.fullmatch(r"customBytes\‍[(\d+)\.\.(\d+)\‍]", source)
260 if span:
261 return int(span.group(1)), int(span.group(2))
262 raise ValueError(f"unsupported state source expression: {source}")

265def state_field_code(field, index):
266 name = field["name"]
267 field_type = field["type"]
268 start, end = parse_source(field["source"])
269 prefix = f'result += ",\\\"{name}\\\":";'
270 if field_type in ("uint8", "boolean"):
271 return f'{prefix}\n result += String((int)description.customBytes[{start}]);'
272 if field_type == "float32":
273 return f'''{prefix}
274 float value{index} = 0.0f;
275 memcpy(&value{index}, &description.customBytes[{start}], sizeof(value{index}));
276 result += String(value{index});'''
277 if field_type == "hex-color":
278 return f'''{prefix}
279 result += "\\\"#" + rgbHex(description.customBytes[{start}], description.customBytes[{start + 1}], description.customBytes[{start + 2}]) + "\\\"";'''
280 array_match = re.fullmatch(r"(boolean|uint8|hex-color)\‍[(\d+)\‍]", field_type)
281 if array_match:
282 element_type, count_text = array_match.groups()
283 count = int(count_text)
284 stride = 3 if element_type == "hex-color" else 1
285 value = (
286 f'"\\\"#" + rgbHex(description.customBytes[{start} + item * 3], '
287 f'description.customBytes[{start + 1} + item * 3], description.customBytes[{start + 2} + item * 3]) + "\\\""'
