292bool ConfigProvider::saveRamMirrorToNvm()
293{
294 bool retVal = false;
295
296 /* Get size of : Device configuration + all the datablocks to save merged data buffer into NvM */
297 uint16_t mergedBufferSize = totalNvmSize;
298
299 /* Create merged buffer */
300 uint8_t* mergedBuffer = (uint8_t*) malloc (mergedBufferSize);
301
302 /* Dynamic buffer allocation succeeded */
303 if(mergedBuffer != nullptr) {
304 /* Copy device config to the buffer */
305 memcpy(mergedBuffer, &configRamMirror, configRamMirror.getSize());
306
307 /* Copy datablocks to the buffer */
308 memcpy(
309 mergedBuffer + configRamMirror.getSize(),
310 dataBlocksRamMirror->data,
311 PersistentDatablockID::e_NUMBER_OF_PERSISTENT_BLOCKS * PersistentDataBlock::getSize()
312 );
313
314 /* Try to save entire mergedBuffer into persistent memory */
315 if(PersistentMemoryAccess::saveData(mergedBuffer, mergedBufferSize))
316 {
317 /* Update data container signals */
318 updateNodeConfigurationSignal();
319 retVal = true;
320 }else
321 {
322 Logger::log("Saving config failed!");
323 }
324
325 /* Release merged buffer resources */
326 free(mergedBuffer);
327 }
328
329 return retVal;
