116void RFManager::evaluateButtonEvent()
117{
118 uint64_t systemDigitalEventId = currentReceptionEvent.buttonUniqueNumber;
119
120 // Check if button was clicked or long pressed
121 if ((currentReceptionEvent.endTime - currentReceptionEvent.startTime) < 250)
122 {
123 // Single click event
124 systemDigitalEventId += 1; // offset for single click
125 Logger::log("RFManager:// Single click event detected for button: " + String((unsigned long)currentReceptionEvent.buttonUniqueNumber));
126 }
127 else
128 {
129 // Long press event
130 systemDigitalEventId += 2; // offset for long press
131 Logger::log("RFManager:// Long press event detected for button: " + String((unsigned long)currentReceptionEvent.buttonUniqueNumber));
132 }
133
134 std::any_cast<std::function<void(uint64_t)>>(DataContainer::getSignalValue(CBK_FIRE_DIGITAL_EVENT))(systemDigitalEventId);
135}

137void RFManager::handleIdleState()
138{
139 static bool lastReceptionState = false;
140 static uint8_t completionCounter = 0;
141
142 if ((millis() - lastEventCompletionTime) < TIME_BETWEEN_EVENTS_MS)
143 {
144 // Prevent handling new events too quickly after last one
145 return;
146 }
147
148 RFButtonInfo receivedButton = receiveButtonSignal();
149 if (receivedButton.buttonUniqueNumber != 0)
150 {
151
152 // Is one of known buttons?
153 if (connectedButtons.find(receivedButton) != connectedButtons.end())
154 {
155 // First click detected
156 if (currentReceptionEvent.buttonUniqueNumber == 0)
157 {
158 currentReceptionEvent.buttonUniqueNumber = receivedButton.buttonUniqueNumber;
159 currentReceptionEvent.startTime = millis();
160 }
161 else if (currentReceptionEvent.buttonUniqueNumber != receivedButton.buttonUniqueNumber)
162 {
163 // Different button received, reset current event
164 resetReceptionEvent();
165 currentReceptionEvent.buttonUniqueNumber = receivedButton.buttonUniqueNumber;
166 currentReceptionEvent.startTime = millis();
167 }
168
169 lastReceptionState = true;
170 }
171 }
172 else
173 {
174
175 // Signal stopped, increase amount of clicks if within time window
176 if (currentReceptionEvent.buttonUniqueNumber != 0)
177 {
178 if (lastReceptionState)
179 {
180 currentReceptionEvent.endTime = millis();
181 lastReceptionState = false;
182 }
183 else
184 {
185 completionCounter++;
186 }
187
188 if (completionCounter >= 5)
189 {
190 // calculate duration of the press
191 long pressDuration = currentReceptionEvent.endTime - currentReceptionEvent.startTime;
192 Logger::log("RFManager:// Button event ended. Duration: " + String((long)pressDuration) + " ms");
193
194 uint64_t systemDigitalEventId = currentReceptionEvent.buttonUniqueNumber;
195 if (pressDuration < TIME_FOR_LONG_PRESS_MS)
196 {
197 // Short press, wait for possible next press
198 systemDigitalEventId += 1;
199 }
200 else
201 {
202 // Long press, evaluate immediately
203 systemDigitalEventId += 2;
204 }
205
206 resetReceptionEvent();
207 completionCounter = 0;
208 lastEventCompletionTime = millis();
209 std::any_cast<std::function<void(uint64_t)>>(DataContainer::getSignalValue(CBK_FIRE_DIGITAL_EVENT))(systemDigitalEventId);
210 }
211 }
212 }
