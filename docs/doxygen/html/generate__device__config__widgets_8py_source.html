46def validate_field(path, field):

56def field_initializer(field):
57 validate_field("device description", field)
58 options = field.get("options", [])
59 if len(options) > 2:
60 raise ValueError(f"{field['name']}: the generated renderer supports at most two select options")
61
62 option_values = [str(option["value"]) for option in options]
63 option_labels = [option["label"] for option in options]
64 option_values.extend([""] * (2 - len(option_values)))
65 option_labels.extend([""] * (2 - len(option_labels)))
66 return "{{{key}, {label}, {html_type}, {offset}, {size}, {minimum}, {maximum}, {{{option_values}}}, {{{option_labels}}}, {option_count}, {array_length}, {display_scale}, {default_value}}}".format(
67 key=cpp_string(field["fieldId"]),
68 label=cpp_string(field["label"]),
69 html_type=cpp_string(field["htmlType"]),
70 offset=int(field["offset"]),
71 size=int(field["size"]),
72 minimum=int(field.get("minimum", 0)),
73 maximum=int(field.get("maximum", 255)),
74 option_values=", ".join(cpp_string(value) for value in option_values),

83def device_initializer(description):
84 device_type = description["deviceType"]
85 fields = [field for field in description["configuration"]["customBytes"] if field.get("widget") is not None]
86 if len(fields) > 8:
87 raise ValueError(f"{device_type['name']}: a generated device type supports at most eight configuration fields")
88
89 fields.extend([None] * (8 - len(fields)))
90 field_rows = [field_initializer(field) if field else '{"", "", "", 0, 0, 0, 0, {"", ""}, {"", ""}, 0, 0, 1, 0}' for field in fields]
91 http_config = device_type.get("httpConfig", {})
92 display_name = http_config.get("displayName", device_type["name"])
93 initializer = "{{{type_id}, {name}, {enum_symbol}, {{{fields}}}, {field_count}}}".format(
94 type_id=int(device_type["enumValue"]),
95 name=cpp_string(display_name),
96 enum_symbol=cpp_string(device_type["enumSymbol"]),
97 fields=",\n ".join(field_rows),
98 field_count=sum(field is not None for field in fields),
99 )
100 build_guard = description["implementation"].get("buildGuard")
101 if build_guard:
102 return f"#ifdef {build_guard}\n {initializer},\n #endif"
103 return initializer + ","

257def generate():
