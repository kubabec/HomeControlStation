69void HTTPAsyncRequestHandler::createServiceCall()
70{
71 ServiceParameters_set1 set1;
72 ServiceParameters_set2 set2;
73 ServiceParameters_set3 set3;
74
75 ServiceRequestErrorCode serviceCallStatus = SERV_GENERAL_FAILURE;
76
77 switch (currentRequest.requestData[SERVICE_OVERLOADING_FUNCTION_INDEX])
78 {
79 /* No parameters service */
80 case serviceCall_NoParams:
81 serviceCallStatus = (std::any_cast<DeviceServicesAPI>(DataContainer::getSignalValue(SIG_DEVICE_SERVICES))).serviceCall_NoParams(currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL], (DeviceServicesType)currentRequest.requestData[SERVICE_NAME_INDEX]);
82 break;
83
84 /* Service with 1 parameter set */
85 case serviceCall_1:
86 memcpy(&set1, &currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL + 1], sizeof(ServiceParameters_set1));
87 serviceCallStatus = (std::any_cast<DeviceServicesAPI>(DataContainer::getSignalValue(SIG_DEVICE_SERVICES))).serviceCall_set1(currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL], (DeviceServicesType)currentRequest.requestData[SERVICE_NAME_INDEX], set1);
88 break;
89
90 case serviceCall_2:
91 memcpy(&set2, &currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL + 1], sizeof(ServiceParameters_set2));
92 serviceCallStatus = (std::any_cast<DeviceServicesAPI>(DataContainer::getSignalValue(SIG_DEVICE_SERVICES))).serviceCall_set2(currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL], (DeviceServicesType)currentRequest.requestData[SERVICE_NAME_INDEX], set2);
93 break;
94
95 case serviceCall_3:
96 /* Fulfill device service call parameters based on HTTP async request data */
97 set3.buff = &(currentRequest.requestData[DYNAMIC_REQUEST_START_OF_DATA_IDX]);
98 set3.direction = currentRequest.requestData[DYNAMIC_REQUEST_DIRECTION_IDX];
99 set3.additionalParam = currentRequest.requestData[DYNAMIC_REQUEST_ADDITIONAL_PARAM_IDX];
100 memcpy(&set3.size, &currentRequest.requestData[DYNAMIC_REQUEST_MEMORY_LENGTH_IDX], sizeof(set3.size));
101
102 serviceCallStatus = (std::any_cast<DeviceServicesAPI>(DataContainer::getSignalValue(SIG_DEVICE_SERVICES))).serviceCall_set3(currentRequest.requestData[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL], (DeviceServicesType)currentRequest.requestData[SERVICE_NAME_INDEX], set3);
103
104 break;
105
106 default:
107 break;
108 }
109
110 /* we must check the status of service call */
111 switch (serviceCallStatus)
112 {
113 case SERV_SUCCESS:
114 /* Request completed successfully */
115 Logger::log("Service request processing finished.");
116 currentRequest.state = ASYNC_REQUEST_COMPLETED;
117 break;
118 case SERV_PENDING:
119 /* We are waiting for the response, service must be polled */
120 // Logger::log("HTTPAsyncRequestHandler: Request call returned PENDING");
121 currentRequest.state = ASYNC_REQUEST_PROCESSING;
122 break;
123 case SERV_BUSY:
124 /* Trying to send request, but actually it is not possible */
125 Logger::log("HTTPAsyncRequestHandler: Request call returned BUSY");
126 /* Dropping the request */
127 currentRequest.state = ASYNC_REQUEST_COMPLETED;
128 /* This will just do not return any valid JSON response */
129 currentRequest.type = ASYNC_TYPE_INVALID;
130 break;
131
132 case SERV_NOT_SUPPORTED:
133 /* Device does not support this service call */
134 Logger::log("HTTPAsyncRequestHandler: Request call returned NOT_SUPPORTED");
135 currentRequest.state = ASYNC_REQUEST_COMPLETED;
136 currentRequest.type = ASYNC_TYPE_INVALID;
137 break;
138
139 default:
140 Logger::log("HTTPAsyncRequestHandler: Request call returned GENERAL_FAILURE, error code: " + String((int)serviceCallStatus));
141 currentRequest.state = ASYNC_REQUEST_COMPLETED;
142 currentRequest.type = ASYNC_TYPE_INVALID;
143 break;
144 }
