158void WiFiAdapter::manualStatusCheck()
159{
160 unsigned long now = millis();
161 lastStatusCheck = now;
162
163 bool isNowConnected = (WiFi.isConnected() && WiFi.status() == WL_CONNECTED);
164
165 if (isNowConnected && !isConnectedFlag)
166 {
167 // event nie został złapany (może wersja core), więc synchronizujemy
168 Logger::log("WiFiAdapter://Connected (Manual check).");
169 updateDataContainerIpAddress();
170
171 isConnectedFlag = true;
172 currentStatus = CONNECTED_TO_NETWORK;
173 nextReconnectAt = 0;
174 }
175 else if (!isNowConnected && isConnectedFlag)
176 {
177 // utrata połączenia wykryta przez polling
178 Logger::log("WiFi: Connection lost");
179 isConnectedFlag = false;
180 currentStatus = DISCONNECTED_FROM_NETWORK;
181 nextReconnectAt = now + reconnectDelayMs;
182 }
183}
