332int OnOffDevice::mapBrightness(int brightness)
333{
334 // if(brightness < 95){
335 // // Need to map that 0-99 percent must be PWM in range between 0-60%
336 // brightness = (brightness * 70) / 100; // Map to 0-60%
337 // }
338
339 if (activeLow)

341 // For devices activated by LOW (0V = ON)
