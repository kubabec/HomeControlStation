24static uint8_t nextMappingId(const std::vector<std::pair<uint64_t, DigitalEvent::Event>> &mappings)
25{
26 for (uint16_t candidate = 1; candidate <= 255; ++candidate)
27 {
28 if (candidate == LEGACY_ROOM_TYPE || candidate == LEGACY_DEVICE_TYPE) continue;
29 bool used = false;
30 for (const auto &mapping : mappings)
31 {
32 if (mapping.second.mappingId == candidate)
33 {
34 used = true;
35 break;
36 }
37 }
38 if (!used) return static_cast<uint8_t>(candidate);
39 }
40 return 0;
41}

290void DigitalEventReceiver::receiveUDP(MessageUDP &msg)
291{
292 /* Received UDP Message */
293 // Process the received digital Event message here
294 if (msg.getId() == DIGITAL_EVENT_FIRED_MSG_ID || msg.getId() == DIGITAL_EVENT_FIRED_WITH_SOURCE_MSG_ID)
295 {
296 std::vector<uint8_t> &payload = msg.getPayload();
297 const size_t legacyPayloadSize = sizeof(uint8_t) + sizeof(uint64_t);
298 const bool legacyMessage = msg.getId() == DIGITAL_EVENT_FIRED_MSG_ID;
299 if ((legacyMessage && payload.size() == legacyPayloadSize) ||
300 (!legacyMessage && payload.size() > legacyPayloadSize &&
301 payload.size() <= legacyPayloadSize + DIGITAL_EVENT_SOURCE_MAX_LENGTH))
302 {
303 uint8_t transmissionIdentfier = 0x00;
304 memcpy(&transmissionIdentfier, &(payload.at(0)), sizeof(transmissionIdentfier));
305 uint64_t triggeredEvent = 0;
306 memcpy(&triggeredEvent, &(payload.at(1)), sizeof(triggeredEvent));
307
308 const auto &sender = msg.getIPAddress();
309 const uint32_t senderKey = (static_cast<uint32_t>(sender.octet1) << 24) |
310 (static_cast<uint32_t>(sender.octet2) << 16) |
311 (static_cast<uint32_t>(sender.octet3) << 8) |
312 sender.octet4;
313 auto senderState = std::find_if(receivedTransmissionIds.begin(), receivedTransmissionIds.end(),
314 [senderKey](const auto &entry) { return entry.first == senderKey; });
315 if (senderState == receivedTransmissionIds.end() || senderState->second != transmissionIdentfier)
316 {
317 String source = "Legacy remote";
318 if (!legacyMessage)
319 {
320 source = "";
321 for (size_t i = legacyPayloadSize; i < payload.size(); ++i) source += static_cast<char>(payload[i]);
322 }
323 Logger::log("DigitalEventReceiver:// Received digital event ID: " + String((unsigned long long)triggeredEvent) + " from " + source + " with transmission ID: " + String((int)transmissionIdentfier));
324 if (senderState == receivedTransmissionIds.end())
325 {
326 if (receivedTransmissionIds.size() >= 32) receivedTransmissionIds.erase(receivedTransmissionIds.begin());
327 receivedTransmissionIds.push_back({senderKey, transmissionIdentfier});
328 }
329 else
330 {
331 senderState->second = transmissionIdentfier;
332 }
333 fireEvent(triggeredEvent, source);
334
335 // Send back the handling confirmation
336 MessageUDP confirmationMessage(DIGITAL_EVENT_CONFIRMED_MSG_ID, msg.getIPAddress(), 9001);
337 confirmationMessage.pushData((uint8_t *)&triggeredEvent, sizeof(uint64_t));
338 NetworkDriver::send(confirmationMessage);
339 }
340 }
341 else
342 {
343 Logger::log("DigitalEventReceiver:// Invalid length of received DigitalEvent message");
344 }
345 }
