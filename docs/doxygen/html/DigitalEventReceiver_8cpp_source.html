261void DigitalEventReceiver::receiveUDP(MessageUDP &msg)
262{
263 /* Received UDP Message */
264 // Process the received digital Event message here
265 if (msg.getId() == DIGITAL_EVENT_FIRED_MSG_ID)
266 {
267 std::vector<uint8_t> &payload = msg.getPayload();
268 if (payload.size() == (sizeof(uint8_t) + sizeof(uint64_t)))
269 {
270 uint8_t transmissionIdentfier = 0x00;
271 memcpy(&transmissionIdentfier, &(payload.at(0)), sizeof(transmissionIdentfier));
272 uint64_t triggeredEvent = 0;
273 memcpy(&triggeredEvent, &(payload.at(1)), sizeof(triggeredEvent));
274
275 // Protection against repeated request fake activation
276 if (transmissionIdentfier != lastReceivedTransmissionId)
277 {
278 Logger::log("DigitalEventReceiver:// Received DIGITAL_EVENT_FIRED_MSG_ID for event ID: " + String((unsigned long long)triggeredEvent) + " with transmission ID: " + String((int)transmissionIdentfier));
279 lastReceivedTransmissionId = transmissionIdentfier;
280 fireEvent(triggeredEvent);
281
282 // Send back the handling confirmation
283 MessageUDP confirmationMessage(DIGITAL_EVENT_CONFIRMED_MSG_ID, msg.getIPAddress(), 9001);
284 confirmationMessage.pushData((uint8_t *)&triggeredEvent, sizeof(uint64_t));
285 NetworkDriver::send(confirmationMessage);
286 }
287 }

290 Logger::log("DigitalEventReceiver:// Invalid length of received DigitalEvent message");
291 }
