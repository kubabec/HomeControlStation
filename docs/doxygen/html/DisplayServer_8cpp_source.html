66void DisplayServer::handleNewDisplay(MessageUDP &msg){
67 // Check if display is already know by unpacking data from message as DisplayInfo and add to knownDisplays if not present
68 DisplayInfo newDisplayInfo;
69 std::vector<uint8_t> &payload = msg.getPayload();
70 if(payload.size() == sizeof(DisplayInfo)){
