71 enum EventType

73 SINGLE_CLICK,

75 LONG_PRESS
76 };
77
78 if (!buttonActionsQueue.empty())
79 {
80 bool isPressed = activeHigh ? (buttonActionsQueue.front().first == 1) : (buttonActionsQueue.front().first == 0);
81 // Button clicked - event started here!
82 if (!eventStarted && isPressed)
83 {
84 eventStartTime = buttonActionsQueue.front().second;
85 eventStarted = true;
86 }
87 else if (eventStarted && !isPressed)
88 { // Button released - event either finished or midtime release happened
89
90 eventEndTime = buttonActionsQueue.front().second;
91 }
92 else if (eventStarted && isPressed)
93 {
94 // Pressed more than one time
95 doubleClick = true;
96 eventForceFinished = true;
97 }
98
99 buttonActionsQueue.pop();
100 }
101
102 if (eventStarted)
103 {
104 // Shall event results be evaluated?
105 if (eventForceFinished || millis() - eventStartTime >= eventDurationTimeMs)
106 {
107 // Event completed
108 eventStarted = false;
109
110 // Evaluate results
111 EventType event = SINGLE_CLICK;
112
113 if (doubleClick)
114 {
115 event = DOUBLE_CLICK;
116 // Logger::log("DOUBLE CLICK, eventStartTime:" + String((int)eventStartTime) + " , eventEndTime: " + String((int)eventEndTime) + " ");
117 }
118 else
119 {
120 // Buton was not released at all
121 if (eventEndTime <= eventStartTime)
122 {
123 event = LONG_PRESS;
124 // Logger::log("LONG PRESS 1, eventStartTime:" + String((int)eventStartTime) + " , eventEndTime: " + String((int)eventEndTime) + " ");
125 }
126 else
127 {
128 // Button was released during event, check the event type
129 if ((float)(eventEndTime - eventStartTime) > (3.f * (((float)eventDurationTimeMs) / 4.f)))
130 {
131 // Touch longer than half of cycle
132 event = LONG_PRESS;
133 // Logger::log("LONG PRESS 2, eventStartTime:" + String((int)eventStartTime) + " , eventEndTime: " + String((int)eventEndTime) + " ");
134 }
135 else
136 {
137 event = SINGLE_CLICK;
138 // Logger::log("SINGLE CLICK, eventStartTime:" + String((int)eventStartTime) + " , eventEndTime: " + String((int)eventEndTime) + " ");
139 }
140 }
141 }
142
143 switch (event)
144 {
145 case SINGLE_CLICK:
146 if(localDeviceTriggerFunction != nullptr){
147 Logger::log("Triggering local device");
148 localDeviceTriggerFunction(localDeviceIdToTrigger);
149 }else {
150 if(eventTriggerFunction != nullptr){
151 Logger::log("Triggering remote device");
152 eventTriggerFunction(optionalSinglePressEventId);
153 }
154 }
155 break;
156 case DOUBLE_CLICK:
157 if(eventTriggerFunction != nullptr){

159 eventTriggerFunction(doubleClickEventId);
160 }
161 break;
162 case LONG_PRESS:
163 if(eventTriggerFunction != nullptr){

165 eventTriggerFunction(longPressEventId);
166 }
167 break;

170 break;
171 }
172
173 // Clean up
174 eventEndTime = 0;
175 eventStartTime = 0;
176 eventStarted = false;
177 doubleClick = false;
178 eventForceFinished = false;
179 }
