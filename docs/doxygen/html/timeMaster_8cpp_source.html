24void TimeMaster::init()
25{
26 Logger::log("DeviceProvider init ...");
27 RtcTime startupTime;
28 DataContainer::setSignalValue(SIG_STARTUP_TIME, static_cast<RtcTime>(startupTime));
29
30 timeClient.begin();
31 setTimeZone(1); // Ustawienie strefy czasowej na GMT+1
32 DataContainer::setSignalValue(CBK_GET_CURRENT_TIME, static_cast<std::function<RtcTime()>>(TimeMaster::getRtcTime));
33
34 // Próba synchronizacji z NTP

36 {
37 updateNtpVariables();
38 Logger::log("Synchronized with NTP!");
39 DataContainer::setSignalValue(SIG_STARTUP_TIME, static_cast<RtcTime>(getRtcTime()));
40 startupTimeInitialized = true; // Czas startowy został zainicjalizowany

63void TimeMaster::cyclic()
64{
65
66 if (!wasNtpEverSynced && initialRetryCount < 60)
67 {
68 if (timeClient.update())
69 {
70 updateNtpVariables();
71 initialRetryCount = 0; // Resetujemy licznik prób synchronizacji
72
73 // Do only once after first successful NTP sync
74 if (!startupTimeInitialized)
75 {
76 DataContainer::setSignalValue(SIG_STARTUP_TIME, static_cast<RtcTime>(getRtcTime()));
77 startupTimeInitialized = true; // Czas startowy został zainicjalizowany
78
79 UserInterfaceNotification notif;
80 notif.title = "SYSTEM RUNNING";
81 notif.body = "Welcome to Home Control Station!";
82 notif.type = UserInterfaceNotification::INFO;
83 std::any_cast<UINotificationsControlAPI>(DataContainer::getSignalValue(SIG_UI_NOTIFICATIONS_CONTROL)).createNotification(notif);
84 }
85 }
86 else
87 {
88 initialRetryCount++;
89 }
90 }
91
92 unsigned long now = millis();
93
94 if (now - lastUpdateTime >= NTP_RESYNC_TIME)
95 {
96 Logger::log("NTP resyncing ...");
97 if (timeClient.update())
98 {
99 updateNtpVariables();
100 timeSyncFailureCount = 0;
101 }
102 else
103 {
104 // Błąd synchronizacji NTP - tryb freerunning
105 timeSyncFailureCount++;
106
107 // Try to reconnect the WiFi when 3x times time sync failed
108 if (timeSyncFailureCount == 3)
109 {
110 timeSyncFailureCount = 0;
111 {
112 std::any localAny{DataContainer::getSignalValue(CBK_RECONNECT_WIFI)};
113 if (auto p = std::any_cast<std::function<void()>>(&localAny))
114 {
115 (*p)();
116 }
117 else
118 {
119 }
120 }
121 }
122
123 ntpAvailable = false;
124 lastUpdateTime = now - HALF_NTP_RESYNC_TIME;
125 }
126 }
127}

129RtcTime TimeMaster::getRtcTime()
130{
131 time_t rawTime = TimeMaster::getEpochTime(); // Pobranie czasu Unix
132 struct tm *timeInfo = localtime(&rawTime); // Konwersja na czas UTC
133
134 RtcTime rtcTime;
135 rtcTime.sec = timeInfo->tm_sec;
136 rtcTime.min = timeInfo->tm_min;
137 rtcTime.hour = timeInfo->tm_hour;
138 rtcTime.mday = timeInfo->tm_mday;
139 rtcTime.mon = timeInfo->tm_mon + 1; // Miesiące są indeksowane od 0
140 rtcTime.year = timeInfo->tm_year + 1900; // Rok od 1900
141 rtcTime.wday = timeInfo->tm_wday;
142 rtcTime.yday = timeInfo->tm_yday;
143 rtcTime.isdst = timeInfo->tm_isdst;
144
145 return rtcTime; // Zwracamy strukturę RtcTime
146}
