149void NetworkDriver::cyclic()
150{
151 /******* WiFi connection section ******/
152 // Update WiFi Adapter to be updated with connection lost status
153 WiFiAdapter::task();
154
155 /******* UDP communication section ******/
156 // Update UDP task to be able to receive UDP packets
157 UDPAdapter::task();
158
159 if (UDPAdapter::sendingAllowed())
160 {
161 static uint8_t retryCount = 0;
162 // Send one pending packet per cyclic
163 if (pendingToSendPackets.size() > 0)
164 {
165 // Send one pending packet per cyclic
166 if (UDPAdapter::send(pendingToSendPackets.front()))
167 {
168 pendingToSendPackets.pop();
169 retryCount = 0;
170 }
171 else
172 {
173 Logger::log("NetworkDriver:// Sending UDP packet (MsgId : " + String((int)pendingToSendPackets.front().getId()) + ") to host " + pendingToSendPackets.front().getIPAddress().toString() + " failed.");
174 retryCount++;
175 if (retryCount > 5)
176 {
177 pendingToSendPackets.pop();
178 retryCount = 0;
179 }
180 }
181 }
182 }
183
184 // Update OTA task to be able to process OTA requests
185 OTA::cyclic();
