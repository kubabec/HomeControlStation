57/**

61class SingleWaveAnimation : public ILiveAnimation {

64
65 SingleWaveAnimation(uint16_t width, Direction dir, float dimFactor = 0.3f)

79 if (!running) return;
80 // dim and preserve zeros

87 mainStrip[i].g = uint8_t(orig.g * dimFactor);
88 mainStrip[i].b = uint8_t(orig.b * dimFactor);
89 }
90 }
91 // overlay moving block
92 for (uint16_t i = 0; i < width; ++i) {
93 int idx = (dir == Direction::LeftToRight ? pos + i : pos - i);
94 if (idx < 0 || idx >= int(ledsCount)) continue;
95 // only apply to originally non-zero
96 const auto &orig = savedState[idx];
97 if (orig.r || orig.g || orig.b) {
98 mainStrip[idx] = averageColor;
99 }
100 }

102 }
103

105 void onStop() override {}

116 uint32_t countNonZero = 0;

118 if (c.r || c.g || c.b) {
119 sumR += c.r;
120 sumG += c.g;
121 sumB += c.b;
122 ++countNonZero;
123 }
124 }
125 if (countNonZero == 0) countNonZero = 1;
126 averageColor.r = uint8_t(sumR / countNonZero);
127 averageColor.g = uint8_t(sumG / countNonZero);
128 averageColor.b = uint8_t(sumB / countNonZero);

134 dir = Direction::RightToLeft;
135 pos = ledsCount - width;
136 }
137 } else {

139 dir = Direction::LeftToRight;
140 pos = 0;
141 }
142 }

181 // smooth wave: distance-based blend
182 for (uint16_t i = 0; i < ledsCount; ++i) {

184 float t = 1.0f - std::abs(dist) / float(width);

187 // apply only if orig non-zero
188 const auto &orig = savedState[i];
189 if (orig.r || orig.g || orig.b) {
190 uint8_t dr = mainStrip[i].r;
191 uint8_t dg = mainStrip[i].g;
192 uint8_t db = mainStrip[i].b;
193 mainStrip[i].r = uint8_t(dr * (1 - factor) + averageColor.r * factor);
194 mainStrip[i].g = uint8_t(dg * (1 - factor) + averageColor.g * factor);
195 mainStrip[i].b = uint8_t(db * (1 - factor) + averageColor.b * factor);
196 }
197 }
198 advance();
199 }

249/**
250 * @class ComplexSequenceAnimation
251 * @brief Implements the Complex Sequence Animation behavior.
252 */
253class ComplexSequenceAnimation : public ILiveAnimation {
254public:
255 Config cfg;
256
257 ComplexSequenceAnimation(const Config& c = Config()) : cfg(c) {}
258 ~ComplexSequenceAnimation() override = default;

261 ILiveAnimation::initialize(stripPtr, count);
262 avgColor = computeAvg();
263 }

265 void onStart() override {
266 // reset all state
267 state = State::SplitFade;
268 bursts.clear();
269 burstsIssued = 0;
270 splitStep = 0;
271 collapseStep = 0;
272 slidePos = 0;
273 slideForward = true;

276 }
277
278 void process() override {
279 if (!running) return;
280 switch (state) {
281 case State::SplitFade: processSplitFade(); break;
282 case State::RandomCycle: processRandomCycle(); break;
283 case State::Slide: processSlide(); break;
284 case State::Collapse: processCollapse(); break;
285 }
286 }
287
288private:
289 enum class State { SplitFade, RandomCycle, Slide, Collapse } state;
290
291 // SplitFade / Collapse
292 int centerLeft = 0;
293 int centerRight = 0;
294 uint8_t splitStep = 0;
295 uint8_t collapseStep = 0;
296
297 // RandomCycle
298/**
299 * @class Burst
300 * @brief Represents the Burst data structure.
301 */
302 struct Burst { int idx; uint8_t step; uint16_t hold; bool held; };
303 std::vector<Burst> bursts;
304 uint8_t burstsIssued = 0;
305
306 // Slide
307 uint16_t slidePos = 0;
308 bool slideForward= true;
309
310 // Common
311 LedColor avgColor;
312
313 // 1) SplitFade: fade out from center outward
314 void processSplitFade() {
315 std::memcpy(mainStrip, savedState.data(), ledsCount * sizeof(LedColor));
316 float t = float(splitStep) / float(cfg.fadeSteps);
317 t = std::min(t, 1.0f);
318 int left = centerLeft - splitStep;
319 int right = centerRight + splitStep;
320 for (int i = left; i <= right; ++i) {
321 if (i >= 0 && i < int(ledsCount)) {
322 mainStrip[i].r = uint8_t(savedState[i].r * (1 - t));
323 mainStrip[i].g = uint8_t(savedState[i].g * (1 - t));
324 mainStrip[i].b = uint8_t(savedState[i].b * (1 - t));
325 }
326 }
327 if (++splitStep > std::max(centerLeft, ledsCount - 1 - centerRight)) {
328 state = State::RandomCycle;
329 }
330 }
331
332 // 2) RandomCycle: bursts with hold at peak, ~10% of leds each spawn
333 void processRandomCycle() {
334 // clear
335 for (uint16_t i = 0; i < ledsCount; ++i)
336 mainStrip[i] = LedColor(0,0,0);
337 // spawn new burst when previous held or none, up to maxBursts
338 if ((bursts.empty() || bursts.back().held) && burstsIssued < cfg.maxBursts) {
339 int burstCount = std::max<int>(1, ledsCount / 10);
340 for (int b = 0; b < burstCount; ++b) {
341 int idx = std::rand() % ledsCount;
342 bursts.push_back({idx, 0, 0, false});
343 }
344 ++burstsIssued;
345 }
346 // update all bursts
347 for (auto it = bursts.begin(); it != bursts.end();) {
348 float t = float(it->step) / float(cfg.fadeSteps - 1);
349 if (!it->held && it->step >= cfg.fadeSteps/2) {
350 if (++(it->hold) < cfg.holdSteps) {
351 mainStrip[it->idx] = savedState[it->idx];
352 ++it;
353 continue;
354 } else {
355 it->held = true;
356 }
357 }
358 float blend = (t <= 0.5f) ? (t * 2.0f) : (2.0f * (1.0f - t));
359 int idx = it->idx;
360 mainStrip[idx].r = uint8_t(savedState[idx].r * blend);
361 mainStrip[idx].g = uint8_t(savedState[idx].g * blend);
362 mainStrip[idx].b = uint8_t(savedState[idx].b * blend);
363 if (++(it->step) >= cfg.fadeSteps) {
364 it = bursts.erase(it);
365 } else {
366 ++it;
367 }
368 }
369 if (burstsIssued >= cfg.maxBursts && bursts.empty()) {
370 // state = State::Slide;
371 // slidePos = 0;
372 // slideForward = true;
373 burstsIssued = 0; // reset for next cycle
374 }
375 }
376
377 // 3) Slide: smooth fade of segment
378 void processSlide() {
379 for (uint16_t i = 0; i < ledsCount; ++i) mainStrip[i] = LedColor(0,0,0);
380 for (uint16_t i = 0; i < cfg.slideWidth; ++i) {
381 int idx = slideForward ? slidePos + i : slidePos - i;
382 if (idx < 0 || idx >= int(ledsCount)) continue;
383 float x = float(i) / float(cfg.slideWidth - 1);
384 float ease = (std::sin((x - 0.5f) * float(M_PI)) + 1.0f) * 0.5f;
385 auto &orig = savedState[idx];
386 mainStrip[idx].r = uint8_t(orig.r * ease + avgColor.r * (1 - ease));
387 mainStrip[idx].g = uint8_t(orig.g * ease + avgColor.g * (1 - ease));
388 mainStrip[idx].b = uint8_t(orig.b * ease + avgColor.b * (1 - ease));
389 }
390 if (slideForward) {
391 if (++slidePos > int(ledsCount - cfg.slideWidth)) {
392 state = State::Collapse;
393 collapseStep = 0;
394 }
395 }
396 }
397
398 // 4) Collapse: grow from edges to center
399 void processCollapse() {
400 std::memcpy(mainStrip, savedState.data(), ledsCount * sizeof(LedColor));
401 int maxSpread = std::max(centerLeft, int(ledsCount - 1 - centerRight));
402 float t = float(collapseStep) / float(maxSpread);
403 t = std::min(t, 1.0f);
404 for (int s = 0; s <= collapseStep; ++s) {
405 int left = centerLeft - s;
406 int right = centerRight + s;
407 if (left >= 0) mainStrip[left] = blendColor(savedState[left], t);
408 if (right < ledsCount) mainStrip[right] = blendColor(savedState[right], t);
409 }
410 if (++collapseStep > maxSpread) {
411 onStart();
412 }
413 }
414
415 // Blend original color by factor
416 LedColor blendColor(const LedColor &orig, float factor) const {
417 return LedColor(
418 uint8_t(orig.r * factor),
419 uint8_t(orig.g * factor),
420 uint8_t(orig.b * factor)
421 );
422 }
423
424 // Compute average non-zero color
425 LedColor computeAvg() const {
426 uint32_t sumR=0, sumG=0, sumB=0, count=0;
427 for (auto &c : savedState) {
428 if (c.r || c.g || c.b) {
429 sumR += c.r;
430 sumG += c.g;
431 sumB += c.b;
432 ++count;
433 }
434 }
435 if (!count) count = 1;
436 return LedColor(
437 uint8_t(sumR/count),
438 uint8_t(sumG/count),
439 uint8_t(sumB/count)
440 );
441 }
