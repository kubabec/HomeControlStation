396void RemoteControlServer::processUDPMessage(MessageUDP &msg)
397{
398
399 // is handshake message ?
400 if (msg.getId() == RESPONSE_NODE_INITIAL_DATA && (currentState == STATE_REQUEST_NODE_INITIAL_DATA || currentState == STATE_KEEP_ALIVE))
401 {
402 handleHandShakeCommunication(msg);
403 }
404
405 if (msg.getId() == RESPONSE_KEEP_ALIVE && currentState == STATE_KEEP_ALIVE)
406 {
407
408 handleSlaveAliveMonitoring(msg);
409 }
410
411 if (msg.getId() == RESPONSE_NODE_DETAILED_DATA_FROM_SPECIFIC_SLAVE && (currentState == STATE_REQUEST_NODE_DETAILED_DATA || currentState == STATE_KEEP_ALIVE))
412 {
413 handleDetailedDataUpdate(msg);
414 }
415
416 /* Process response from the slave */
417 if (msg.getId() == RC_RESPONSE)
418 {
419 /* Process RcResponse */
420 Logger::log("{RCServer} Received RC_RESPONSE message.");
421 processReceivedRcResponse(msg);
422 // Logger::log("<-RC_RESPONSE");
423 }
424
425 if (msg.getId() == DISCOVER_ME_MESSAGE && currentState == STATE_KEEP_ALIVE)
426 {
427 MessageUDP::IPAddr ip = msg.getIPAddress();
428 Logger::log("{RCServer} Received DISCOVER_ME_MESSAGE, sending discovery request to " + ip.toString());
429 MessageUDP msg(REQUEST_NODE_INITIAL_DATA, ip, 9001);
430 NetworkDriver::send(msg);
431 }
