331uint16_t OperatingSystem::calculateRuntimeNodeHash()
332{
333 uint64_t hash = 0;
334
335 // hash += uniqueLifecycleId;
336
337 /* Get configuration */
338 {
339 std::any localAny{DataContainer::getSignalValue(SIG_DEVICE_CONFIGURATION)};
340 if (auto p = std::any_cast<NodeConfiguration>(&localAny))
341 {
342 NodeConfiguration configuration = *p;
343 hash += configuration.isHttpServer;
344 hash += configuration.isRcServer;
345 hash += configuration.networkCredentialsAvailable;
346 hash += configuration.nodeType;
347 for (uint8_t idx = 0; idx < configuration.networkSSID.length(); idx++)
348 {
349 hash += configuration.networkSSID.charAt(idx);
350 }
351 for (uint8_t idx = 0; idx < configuration.networkPassword.length(); idx++)
352 {
353 hash += configuration.networkPassword.charAt(idx);
354 }
355 }
356 else
357 {
358 }
359 }
360
361 /* Get devices data */
362 {
363 std::any localAny{DataContainer::getSignalValue(SIG_DEVICE_COLLECTION)};
364 if (auto p = std::any_cast<std::vector<DeviceDescription>>(&localAny))
365 {
366 std::vector<DeviceDescription> devicesVector = *p;
367 for (auto &device : devicesVector)
368 {
369 hash += device.deviceId;
370 hash += device.macAddress;
371 hash += device.isEnabled;
372 for (uint8_t idx = 0; idx < NUMBER_OF_CUSTOM_BYTES_IN_DESCRIPTION; idx++)
373 {
374 hash += device.customBytes[idx];
375 }
376 for (uint8_t idx = 0; idx < device.deviceName.length(); idx++)
377 {
378 hash += device.deviceName.charAt(idx);
379 }
380 }
381 }
382 else
383 {
384 }
385 }
386
387 // Logger::log("Hash : " + String((int)hash));
388
389 memcpy(&runtimeNodeHash, &hash, sizeof(uint16_t));
390 DataContainer::setSignalValue(SIG_RUNTIME_NODE_HASH, static_cast<uint16_t>(runtimeNodeHash));
391 // Logger::log("New hash : " + String((int)hash));
392
393 return hash;

445void OperatingSystem::detectHwMassEraseRequest()
446{
447 static uint8_t activationTimeCounter = 0;
448 static bool isMassEraseRequestActivated = false;
449
450 if (isMassEraseRequestActivated)
451 { /* keep it first, to execute reset in the next execution to avoid reset when bootloader btn is pressed */
452 if (digitalRead(0) == 1)
453 { /* button was released, reset can be performed */
454 std::any_cast<std::function<void(void)>>(DataContainer::getSignalValue(CBK_MASS_ERASE))();
455 std::any_cast<std::function<void(uint16_t)>>(DataContainer::getSignalValue(CBK_RESET_DEVICE))(1000);
456 }
457 }
458
459 if (digitalRead(0) == 0)
460 { /* Button is pressed */
461 activationTimeCounter++;
462 // // trigger event TESTCODE
463 // std::any_cast<std::function<void(uint64_t)>>(DataContainer::getSignalValue(CBK_FIRE_DIGITAL_EVENT))(5432);
464 }
465 else
466 {
467 activationTimeCounter = 0; /* reset activation timer */
468 }
469
470 if (activationTimeCounter > 3)
471 { /* 5 sec to activate reset */
