104void ExtendedMemoryManager::restoreExtMemoryFromNvm()
105{
106 uint16_t offsetInNvm = 0;
107 for(uint8_t deviceId = 0 ; deviceId < SLOTS_FOR_EXT_MEMORY_BUFFERS; deviceId++){
108 /* Are there any data for this device? */
109 if(extMemoryMetadata.memoryPerDeviceSlotNeeded[deviceId] > 0){
110 /* try to allocate RAM buffer for data */
111 uint8_t* data = (uint8_t*)malloc(extMemoryMetadata.memoryPerDeviceSlotNeeded[deviceId]);
112 if(data != nullptr){
113 /* Read data from NVM to buffer under 'data' pointer */
114 PersistentMemoryAccess::readExtendedMemoryWithOffset(
115 offsetInNvm,
116 data,
117 extMemoryMetadata.memoryPerDeviceSlotNeeded[deviceId]
118 );
119
120 /* Create new ExtMemory entry */
121 ExtMemoryData entry {
122 .ownerDeviceId = deviceId + 1,
123 .length = extMemoryMetadata.memoryPerDeviceSlotNeeded[deviceId],
124 .dataPtr = data
125 };
126
127 extMemoryInUse += entry.length;
128
129 /* Add ExtMemory entry to the container */
130 extMemoryContainer.push_back(entry);
131
132 /* Move offset to next ExtMemory section in NVM */
133 offsetInNvm += extMemoryMetadata.memoryPerDeviceSlotNeeded[deviceId];
134 }else
