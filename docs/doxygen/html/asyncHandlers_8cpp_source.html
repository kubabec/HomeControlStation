87void HomeLightHttpServer::parameterizedHandler_setAdvancedControls(String &request, WiFiClient &client)
88{
89 escapeSpecialCharsInJson(request);
90 request.replace("/setAdvancedControls&", "");
91
92 JsonDocument doc;
93 const DeserializationError error = deserializeJson(doc, request.c_str());
94 if (error != DeserializationError::Code::Ok || !doc["devId"].is<uint16_t>() || !doc["payload"].is<JsonArray>())
95 {
96 Logger::log("Invalid generic advanced-controls request");
97 return;
98 }
99
100 const uint16_t deviceId = doc["devId"].as<uint16_t>();
101 const uint16_t additionalParam = doc["additionalParam"] | 0xFF;
102 if (deviceId > 255 || additionalParam > 255)
103 {
104 Logger::log("Advanced-controls device ID or action is out of range");
105 return;
106 }
107
108 const JsonArray payload = doc["payload"].as<JsonArray>();
109 const uint16_t payloadSize = payload.size();
110 const uint16_t requestSize = DYNAMIC_REQUEST_START_OF_DATA_IDX + payloadSize;
111 if (requestSize >= MAX_PARAM_LENGTH_FOR_ASYNC_REQUEST)
112 {
113 Logger::log("Advanced-controls payload exceeds async request capacity");
114 return;
115 }
116
117 uint8_t *memory = static_cast<uint8_t *>(calloc(requestSize, sizeof(uint8_t)));
118 if (memory == nullptr)
119 {
120 Logger::log("Unable to allocate generic advanced-controls request");
121 return;
122 }
123
124 memory[SERVICE_OVERLOADING_FUNCTION_INDEX] = serviceCall_3;
125 memory[SERVICE_NAME_INDEX] = DEVSERVICE_SET_ADVANCED_CONTROLS;
126 memory[DEVICE_ID_IN_ASYNC_REQUEST_SERVICE_CALL] = static_cast<uint8_t>(deviceId);
127 memcpy(memory + DYNAMIC_REQUEST_MEMORY_LENGTH_IDX, &payloadSize, sizeof(payloadSize));
128 memory[DYNAMIC_REQUEST_DIRECTION_IDX] = e_IN_to_DEVICE;
129 memory[DYNAMIC_REQUEST_ADDITIONAL_PARAM_IDX] = static_cast<uint8_t>(additionalParam);
130 for (uint16_t index = 0; index < payloadSize; ++index)
131 {
132 const int value = payload[index].as<int>();
133 if (!payload[index].is<int>() || value < 0 || value > 255)
134 {
135 Logger::log("Advanced-controls payload contains a non-byte value");
136 free(memory);
137 return;
138 }
139 memory[DYNAMIC_REQUEST_START_OF_DATA_IDX + index] = static_cast<uint8_t>(value);
140 }
141
142 HTTPAsyncRequestHandler::createRequest(ASYNC_TYPE_DEVICE_SERVICE_CALL, memory, requestSize);
143 free(memory);
144}
