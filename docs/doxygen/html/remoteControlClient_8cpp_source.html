90{
91 switch (msg.getId())
92 {

94 // Logger::log("-> Dostałem UDP type RC_REQUEST");

100

102 // Logger::log("-> Dostałem UDP REQUEST_NODE_INITIAL_DATA");

107 break;
108
109 case REQUEST_KEEP_ALIVE:
110 // Logger::log("-> Dostałem UDP REQUEST_KEEP_ALIVE");
111 Logger::log("RemoteControlClient:// Received REQUEST_KEEP_ALIVE message.");
112 updateLastKnownMasterIp(msg.getIPAddress());
113 sendKeepAlive();
114 break;
115
116 case REQUEST_NODE_DETAILED_DATA_FROM_SPECIFIC_SLAVE:
117 // Logger::log("-> Dostałem UDP REQUEST_NODE_DETAILED_DATA_FROM_SPECIFIC_SLAVE");
118 Logger::log("RemoteControlClient:// Received REQUEST_NODE_DETAILED_DATA_FROM_SPECIFIC_SLAVE message.");
119 sendDetailedDataResponse(RESPONSE_NODE_DETAILED_DATA_FROM_SPECIFIC_SLAVE);
120 updateLastKnownMasterIp(msg.getIPAddress());
121 break;
122
123 default:
124 break;
125 }
