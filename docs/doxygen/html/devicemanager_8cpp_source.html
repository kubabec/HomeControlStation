269void DeviceManager::cyclic()
270{
271 static uint32_t lastInternalDescriptionUpdateTime = 0;
272 const uint32_t now = millis();
273
274 for (size_t index = 0; index < devices.size(); ++index)
275 {
276 auto &device = devices[index];
277 const auto *registration = GeneratedDeviceRegistry::find(device->getDeviceType());
278 const uint32_t intervalMs = registration != nullptr ? registration->cycleIntervalMs : 0;
279 if (intervalMs == 0 || static_cast<uint32_t>(now - lastDeviceCycleTimes[index]) >= intervalMs)
280 {
281 lastDeviceCycleTimes[index] = now;
282 device->cyclic();
283 }
284 }
285
286 if (static_cast<uint32_t>(now - lastInternalDescriptionUpdateTime) >= 1000)
287 {
288 updateDeviceDescriptionSignal();
289 lastInternalDescriptionUpdateTime = now;
290 }
291}
