75 sizeof(requestId) + sizeof(targetNodeMAC) + sizeof(targetDeviceId) + sizeof(requestType);
76 const uint8_t dataLengthAfterDynamicPart =
77 sizeof(requestSendCount) + sizeof(crc);
78
79 /* copy first part */
80 uint16_t offset = 0;

88 offset += sizeof(requestType);

93 data.clear();
