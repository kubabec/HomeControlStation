198/**
199 * @brief Runtime representation of a device state and metadata snapshot.
200 *
201 * This structure is used to describe a device over the network and to persist
202 * configuration across reboots.
203 */
204struct DeviceDescription
205{
206 /** Hardware type identifier. */
207 uint8_t deviceType = 255;
208 /** Unique MAC address of the device. */
209 uint64_t macAddress = 0LL;
210 /** Runtime device instance number. */
211 uint8_t deviceId = -1;
212 /** Room or zone index associated with the device. */
213 uint8_t roomId = 255;
214 /** Indicates whether the device is active. */
215 uint8_t isEnabled;
216 /** Free-form bytes reserved for custom metadata. */
217 uint8_t customBytes[NUMBER_OF_CUSTOM_BYTES_IN_DESCRIPTION] = {0x00};
218
219 /** Human-readable device name. */
220 String deviceName;
221
222 /**
223 * @brief Initializes the custom metadata area with zeros.
224 */
225 DeviceDescription()
226 {
227 memset(customBytes, 0x00, NUMBER_OF_CUSTOM_BYTES_IN_DESCRIPTION);
228 }

230 /**
231 * @brief Logs the full device description to the system logger.

233 void print()
234 {
235 char macStr[18];
236 snprintf(macStr, sizeof(macStr), "%02X:%02X:%02X:%02X:%02X:%02X:%02X:%02X",
237 (uint8_t)(macAddress >> 56), (uint8_t)(macAddress >> 48),
238 (uint8_t)(macAddress >> 40), (uint8_t)(macAddress >> 32),
239 (uint8_t)(macAddress >> 24), (uint8_t)(macAddress >> 16),
240 (uint8_t)(macAddress >> 8), (uint8_t)(macAddress));
241 String devTypeStr;
242 switch (deviceType)
243 {
244 case type_ONOFFDEVICE:
245 devTypeStr = "ONOFFDEVICE";
246 break;
247 case type_LED_STRIP:
248 devTypeStr = "LED_STRIP";
249 break;

251 devTypeStr = "TEMP_SENSOR";

254 devTypeStr = "LED_STRIP_SEGMENTED";

265 }

279 /**
280 * @brief Returns the serialized size of the device description payload.
281 * @return Byte length of the persisted description block.
282 */
283 uint16_t getSize()
284 {
285 return (sizeof(deviceType) + sizeof(macAddress) + sizeof(deviceId) + sizeof(roomId) + sizeof(isEnabled) + DEVICE_NAME_MAX_LENGHT + NUMBER_OF_CUSTOM_BYTES_IN_DESCRIPTION);
286 };

288 /**
289 * @brief Serializes the description to a byte buffer.
290 * @param buffer Destination memory block.
291 * @param size Expected size in bytes.
292 * @return True when the payload was encoded successfully.
293 */
294 bool toByteArray(uint8_t *buffer, uint16_t size)
295 {
296 uint16_t offset = 0;
297 if (buffer != nullptr && size == getSize())
298 {
299 memcpy(&buffer[offset], &deviceType, sizeof(deviceType));
300 offset += sizeof(deviceType);
301
302 memcpy(&buffer[offset], &macAddress, sizeof(macAddress));
303 offset += sizeof(macAddress);
304
305 memcpy(&buffer[offset], &deviceId, sizeof(deviceId));
306 offset += sizeof(deviceId);
307
308 memcpy(&buffer[offset], &roomId, sizeof(roomId));
309 offset += sizeof(roomId);
310
311 memcpy(&buffer[offset], &isEnabled, sizeof(isEnabled));
312 offset += sizeof(isEnabled);
313
314 /* prepare empty space string value */
315 memset(&buffer[offset], '\0', DEVICE_NAME_MAX_LENGHT);
316 if (deviceName.length() < DEVICE_NAME_MAX_LENGHT)
317 {
318 deviceName.getBytes(&buffer[offset], DEVICE_NAME_MAX_LENGHT);
319 }
320 else
321 {
322 return false;
323 }
324 offset += DEVICE_NAME_MAX_LENGHT;
325
326 memcpy(&buffer[offset], customBytes, NUMBER_OF_CUSTOM_BYTES_IN_DESCRIPTION);
327
328 return true;
329 }
330 return false;
331 }
